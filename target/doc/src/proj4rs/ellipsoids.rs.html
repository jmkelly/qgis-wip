5/// Ellipsoid flattening may be defined either by
6/// the knowledge of its minor axis or by its reverse flattening
7pub enum FlatteningParam {
8    MinorAxis(f64),
9    InvFlat(f64),
10}
11
12use FlatteningParam::*;
13
14pub struct EllipsoidDefn {
15    pub id: &'static str,
16    pub a: f64,
17    pub rf_or_b: FlatteningParam,
18    //pub comment: &'static str,
19}

138/// Return the ellipse definition
139pub fn find_ellipsoid(name: &str) -> Option<&EllipsoidDefn> {
140    constants::ELLIPSOIDS
141        .iter()
142        .find(|e| e.id.eq_ignore_ascii_case(name))
143        .copied()
144}
