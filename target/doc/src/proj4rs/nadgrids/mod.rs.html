23/// NadGrids
24///
25/// Returned from the sequence
26/// of nadgrids from projstring definition
27#[derive(Debug, Clone)]
28pub struct NadGrids(Vec<GridRef>);
29
30impl PartialEq for NadGrids {
31    fn eq(&self, other: &Self) -> bool {
32        // Don't bother to compare all names
33        self.0.is_empty() && other.0.is_empty()
34    }
35}

37impl NadGrids {
38    pub fn apply_shift(
39        &self,
40        dir: Direction,
41        lam: f64,
42        phi: f64,
43        z: f64,
44    ) -> Result<(f64, f64, f64)> {
45        if self.0.is_empty() {
46            return Ok((lam, phi, z));
47        }
48
49        // Find the correct (root)  grid for an input
50        let mut iter = self.0.iter();
51        let mut candidate = iter.find(|g| g.is_root() && g.matches(lam, phi, z));
52
53        // Check for childs grid
54        if let Some(grid) = candidate {
55            iter.try_fold(grid, |grid, g| {
56                if !g.is_child_of(grid) {
57                    // No more childs, stop with the last candidate
58                    ControlFlow::Break(())
59                } else if g.matches(lam, phi, z) {
60                    // Match, check for childs
61                    candidate.replace(g);
62                    ControlFlow::Continue(g)
63                } else {
64                    // Go next child
65                    ControlFlow::Continue(grid)
66                }
67            });
68        }
69
70        match candidate {
71            Some(g) => g.nad_cvt(dir, lam, phi, z),
72            None => Err(Error::PointOutsideNadShiftArea),
73        }
74    }
75
76    /// Return a list of grids from the catalog
77    pub fn new_grid_transform(names: &str) -> Result<Self> {
78        // Parse the grid list and return an error
79        // if there is any mandatory grid or the list is not terminated by
80        // '@null'
81        let mut v: Vec<GridRef> = vec![];
82
83        match names.split(',').try_for_each(|s| {
84            let s = s.trim();
85            if s == "@null" || s == "null" {
86                // Allow empty list
87                // Mark also the end of parsing
88                ControlFlow::Break(true)
89            } else if let Some(s) = s.strip_prefix('@') {
90                // Optional grid
91                catalog::find_grids(s, &mut v);
92                ControlFlow::Continue(())
93            } else {
94                // Mandatory grid
95                if catalog::find_grids(s, &mut v) {
96                    ControlFlow::Continue(())
97                } else {
98                    ControlFlow::Break(false)
99                }
100            }
101        }) {
102            ControlFlow::Break(true) => Ok(Self(v)),
103            ControlFlow::Break(false) => Err(Error::NadGridNotAvailable),
104            _ => {
105                if v.is_empty() {
106                    Err(Error::NadGridNotAvailable)
107                } else {
108                    Ok(Self(v))
109                }
110            }
111        }
112    }
113
114    pub fn is_empty(&self) -> bool {
115        self.0.is_empty()
116    }
